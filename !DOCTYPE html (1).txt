    function obterCampo(objeto, aliases, padrao = undefined) {
      if (!objeto || typeof objeto !== 'object') return padrao;
      const mapa = new Map(Object.keys(objeto).map(chave => [normalizarChave(chave), objeto[chave]]));
      for (const alias of aliases) {
        const chave = normalizarChave(alias);
        if (mapa.has(chave) && mapa.get(chave) !== null && mapa.get(chave) !== undefined && mapa.get(chave) !== '') return mapa.get(chave);

    function numero(valor, padrao = 0) {
      if (typeof valor === 'number' && Number.isFinite(valor)) return valor;
      if (valor === null || valor === undefined || valor === '') return padrao;
      const texto = String(valor).trim().replace(/\s/g, '').replace(/R\$/gi, '').replace(/%$/, '');
      let convertido;
      if (/^-?\d{1,3}(\.\d{3})+(,\d+)?$/.test(texto)) convertido = Number(texto.replace(/\./g, '').replace(',', '.'));
      else convertido = Number(texto.replace(',', '.'));
      return Number.isFinite(convertido) ? convertido : padrao;
    }

    function dataValida(valor) {
      if (valor instanceof Date && !Number.isNaN(valor.getTime())) return valor;
      if (valor === null || valor === undefined || valor === '') return null;
      if (typeof valor === 'number') {
        const data = valor > 100000000000 ? new Date(valor) : valor > 20000 && valor < 100000 ? new Date(Date.UTC(1899, 11, 30) + valor * 86400000) : null;
        return data && !Number.isNaN(data.getTime()) ? data : null;
      }
      const texto = String(valor).trim();
      const br = texto.match(/^(\d{1,2})[\/-](\d{1,2})[\/-](\d{2,4})/);
      if (br) {
        let ano = Number(br[3]);
        if (ano < 100) ano += ano < 50 ? 2000 : 1900;
        const data = new Date(ano, Number(br[2]) - 1, Number(br[1]));
        return Number.isNaN(data.getTime()) ? null : data;
      }
      const data = new Date(texto);
      return Number.isNaN(data.getTime()) ? null : data;
    }

      const b = Date.UTC(fim.getFullYear(), fim.getMonth(), fim.getDate());
      return Math.max(0, Math.floor((b - a) / 86400000));
    }

    function interpretarBoolSla(valor) {

    function calcularMetricas(lista) {
      const ativos = lista.filter(p => !p.encerrado);

      const projetosComDias = lista.filter(p => p.dias !== null && Number.isFinite(p.dias));
      const sla = slaConhecido.length
        ? (slaConhecido.filter(p => p.sla).length / slaConhecido.length) * 100
        : (projetosComDias.length ? (projetosComDias.filter(p => p.dias <= META_DIAS).length / projetosComDias.length) * 100 : null);
      const diasProjetos = lista.map(p => p.dias).filter(v => v !== null && Number.isFinite(v));
      return {
        ativos: ativos.length,
        encerrados: encerrados.length,
        sla,
        tempoMedio: diasEncerrados.length ? media(diasEncerrados) : null,

    function contarPor(lista, chave) {

    function obterMensal() {
      const candidatos = valorBackend(['mensal','entradasEncerramentos','evolucaoMensal','dadosMensais','movimentacaoMensal']);
      if (Array.isArray(candidatos) && candidatos.length) {
        return candidatos.map(item => ({
          mes: String(obterCampo(item, ['mes','mês','label','periodo','período'], '')),
          entradas: numero(obterCampo(item, ['entradas','entrada','novos','abertos'], 0)),
          encerramentos: numero(obterCampo(item, ['encerramentos','encerrados','fechamentos','concluidos'], 0))
        }));
      }
      const temDatas = todosProjetos.some(p => p.dataEntrada || p.dataFechamento);
      if (!temDatas) return [];
      const agora = new Date();
      const meses = [];
      for (let deslocamento = 11; deslocamento >= 0; deslocamento--) {
        const data = new Date(agora.getFullYear(), agora.getMonth() - deslocamento, 1);
        const chave = `${data.getFullYear()}-${data.getMonth()}`;
        meses.push({chave, mes: `${MESES[data.getMonth()].slice(0,3)}/${String(data.getFullYear()).slice(-2)}`, entradas:0, encerramentos:0});
      }
      const lookup = new Map(meses.map(m => [m.chave,m]));
      projetosFiltrados.forEach(p => {
        if (p.dataEntrada) {
          const item = lookup.get(`${p.dataEntrada.getFullYear()}-${p.dataEntrada.getMonth()}`);
          if (item) item.entradas++;
        }
        if (p.dataFechamento) {
          const item = lookup.get(`${p.dataFechamento.getFullYear()}-${p.dataFechamento.getMonth()}`);
          if (item) item.encerramentos++;
        }
      });
      return meses.map(({mes,entradas,encerramentos}) => ({mes,entradas,encerramentos}));
    }

    function escaparHtml(valor) {
      return String(valor ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');
    }

    function linhaVazia(colunas, texto = 'Nenhum registro encontrado para os filtros selecionados.') {
      return `<tr><td class="empty-row" colspan="${colunas}">${escaparHtml(texto)}</td></tr>`;
    }

    function renderizarTop10() {
      let antigos;
      let vencidos;
      if (todosProjetos.length) {
        antigos = projetosFiltrados.filter(p => !p.encerrado && p.dias !== null).sort((a,b)=>b.dias-a.dias).slice(0,10);
        vencidos = projetosFiltrados.filter(p => p.vencido !== null && p.vencido > 0).sort((a,b)=>b.vencido-a.vencido).slice(0,10);
      } else {
        const top = valorBackend(['top10','projetosAntigos','maisAntigos']);
        const lista = Array.isArray(top) ? top.map(normalizarProjeto) : [];
        antigos = [...lista].sort((a,b)=>(b.dias ?? 0)-(a.dias ?? 0)).slice(0,10);
        vencidos = [...lista].filter(p=>(p.vencido ?? 0)>0).sort((a,b)=>(b.vencido ?? 0)-(a.vencido ?? 0)).slice(0,10);
      }
      $('tbodyAntigos').innerHTML = antigos.length ? antigos.map(p=>`<tr><td class="project-name">${escaparHtml(p.nome)}</td><td>${escaparHtml(p.responsavel)}</td><td class="number-cell">${formatarNumero(p.dias)}</td><td class="number-cell ${p.vencido > 0 ? 'danger-text' : ''}">${p.vencido === null ? '—' : formatarNumero(p.vencido)}</td></tr>`).join('') : linhaVazia(4);
      $('tbodyVencidos').innerHTML = vencidos.length ? vencidos.map(p=>`<tr><td class="project-name">${escaparHtml(p.nome)}</td><td>${escaparHtml(p.responsavel)}</td><td class="number-cell danger-text">${p.vencido === null ? '—' : formatarNumero(p.vencido)}</td></tr>`).join('') : linhaVazia(3, 'Não há projetos com dias vencidos informados.');
    }

    function classeStatus(projeto) {
      if (projeto.encerrado) return 'closed';
      if (projeto.vencido !== null && projeto.vencido > 0) return 'late';
      const status = normalizarChave(projeto.status);
      if (status.includes('pendente') || status.includes('aguardando')) return 'pending';
      return 'active';
    }

    function renderizarTabela() {
      const textoBusca = $('pesquisaProjeto').value.trim().toLocaleLowerCase('pt-BR');
      const lista = projetosFiltrados.filter(p => !textoBusca || p.busca.includes(textoBusca));
      const totalPaginas = Math.max(1, Math.ceil(lista.length / ITENS_POR_PAGINA));
      paginaAtual = Math.min(paginaAtual, totalPaginas);
      const inicio = (paginaAtual - 1) * ITENS_POR_PAGINA;
      const pagina = lista.slice(inicio, inicio + ITENS_POR_PAGINA);
      $('tbodyProjetos').innerHTML = pagina.length ? pagina.map(p=>`<tr>
        <td class="number-cell">${escaparHtml(p.id)}</td>
        <td class="project-name">${escaparHtml(p.nome)}</td>
        <td><span class="status-pill ${classeStatus(p)}">${escaparHtml(p.status)}</span></td>
        <td>${escaparHtml(p.responsavel)}</td>
        <td>${escaparHtml(p.tipo)}</td>
        <td>${escaparHtml(p.fase)}</td>
        <td class="number-cell">${p.dias === null ? '—' : formatarNumero(p.dias)}</td>
        <td class="number-cell ${(p.vencido ?? 0)>0 ? 'danger-text' : ''}">${p.vencido === null ? '—' : formatarNumero(p.vencido)}</td>
      </tr>`).join('') : linhaVazia(8, todosProjetos.length ? 'Nenhum projeto corresponde à pesquisa e aos filtros.' : 'O servidor ainda não retornou a lista completa de projetos.');
      $('contadorTabela').textContent = lista.length ? `Exibindo ${inicio + 1}–${Math.min(inicio + ITENS_POR_PAGINA, lista.length)} de ${lista.length.toLocaleString('pt-BR')} projetos` : 'Nenhum projeto encontrado';
      $('indicadorPagina').textContent = `Página ${paginaAtual} de ${totalPaginas}`;
      $('paginaAnterior').disabled = paginaAtual <= 1;
      $('proximaPagina').disabled = paginaAtual >= totalPaginas;
    }

    function mudarPagina(direcao) {
      paginaAtual += direcao;
      renderizarTabela();
      $('tituloCarteira').scrollIntoView({behavior:'smooth',block:'start'});
    }

    function exportarCSV() {
      if (!projetosFiltrados.length) {
        exibirMensagem('Não há projetos carregados para exportar. Confirme se getDashboardData() retorna a propriedade "projetos" com a carteira completa.', 'info');
        return;
      }
      const cabecalho = ['ID','Projeto','Status','Responsável','Tipo','Fase','Data de entrada','Data de encerramento','Dias','Dias vencidos'];
      const linhas = projetosFiltrados.map(p => [p.id,p.nome,p.status,p.responsavel,p.tipo,p.fase,formatarData(p.dataEntrada),formatarData(p.dataFechamento),p.dias ?? '',p.vencido ?? '']);
      const csv = '\uFEFF' + [cabecalho, ...linhas].map(linha => linha.map(campo => `"${String(campo ?? '').replace(/"/g,'""')}"`).join(';')).join('\r\n');
      const blob = new Blob([csv], {type:'text/csv;charset=utf-8;'});
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      link.download = `carteira_projetos_${new Date().toISOString().slice(0,10)}.csv`;
      document.body.appendChild(link);
      link.click();
      link.remove();
      URL.revokeObjectURL(url);
    }

    function formatarData(data) {
      return data instanceof Date && !Number.isNaN(data.getTime()) ? data.toLocaleDateString('pt-BR') : '';
    }
